cmake_minimum_required(VERSION 3.24)
project(cchp VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "CCHP requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
option(CCHP_ANALYZE "Enable MSVC code analysis" OFF)
option(CCHP_ASAN "Enable AddressSanitizer" OFF)
if(CCHP_ANALYZE AND CCHP_ASAN)
  message(FATAL_ERROR "Use separate analysis and ASan builds")
endif()
foreach(lang C CXX)
  string(REGEX REPLACE "/RTC[1su]" "" CMAKE_${lang}_FLAGS_DEBUG "${CMAKE_${lang}_FLAGS_DEBUG}")
endforeach()
string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
add_compile_options(/W4 /utf-8 /EHa)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(CCHP_ANALYZE)
  add_compile_options(/analyze)
endif()
if(CCHP_ASAN)
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/INCREMENTAL:NO)
endif()
add_library(cchp_core STATIC cchp-api-dll/core.cpp cchp-api-dll/http.cpp cchp-api-dll/client.cpp cchp-api-dll/usage.cpp cchp-api-dll/sse.cpp)
target_include_directories(cchp_core PUBLIC cchp-api-dll)
target_include_directories(cchp_core SYSTEM PUBLIC third_party)
target_link_libraries(cchp_core PUBLIC winhttp rpcrt4)
configure_file(cchp-api-dll/cchp.def cchp.def COPYONLY)
add_library(cchp SHARED cchp-api-dll/abi.cpp cchp-api-dll/dllmain.cpp "${CMAKE_CURRENT_BINARY_DIR}/cchp.def")
target_include_directories(cchp PUBLIC include)
target_compile_definitions(cchp PRIVATE CCHP_BUILD)
target_link_libraries(cchp PRIVATE cchp_core)
add_executable(cchp_status examples/status.c)
target_link_libraries(cchp_status PRIVATE cchp)
include(CTest)
if(BUILD_TESTING)
  add_library(mock_server STATIC tests/mock_server.cpp)
  target_link_libraries(mock_server PUBLIC cchp_core ws2_32)
  target_include_directories(mock_server PUBLIC tests)
  add_executable(contract_test tests/contract_test.cpp)
  target_link_libraries(contract_test PRIVATE mock_server)
  add_test(NAME contract COMMAND contract_test)
  add_executable(usage_test tests/usage_test.cpp)
  target_link_libraries(usage_test PRIVATE mock_server)
  add_test(NAME usage COMMAND usage_test)
  add_executable(sse_test tests/sse_test.cpp)
  target_link_libraries(sse_test PRIVATE mock_server cchp)
  add_test(NAME sse COMMAND sse_test)
  set_tests_properties(sse PROPERTIES TIMEOUT 20)
  add_executable(core_test tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE cchp_core)
  add_test(NAME core COMMAND core_test)
  add_executable(abi_test tests/abi_test.c tests/mock_bridge.cpp examples/status.c)
  target_compile_definitions(abi_test PRIVATE CCHP_EXAMPLE_NO_MAIN)
  target_link_libraries(abi_test PRIVATE cchp mock_server)
  add_test(NAME abi COMMAND abi_test)
  add_test(NAME example COMMAND abi_test example)
  set_tests_properties(abi example PROPERTIES TIMEOUT 20)
  if(NOT CCHP_ASAN)
    get_filename_component(msvc_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
      set(cchp_arch x64)
    else()
      set(cchp_arch x86)
    endif()
    add_test(NAME exports COMMAND powershell -NoProfile -ExecutionPolicy RemoteSigned -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check-exports.ps1"
      -Dll $<TARGET_FILE:cchp> -Architecture ${cchp_arch} -Dumpbin "${msvc_bin}/dumpbin.exe")
  endif()
endif()
install(TARGETS cchp RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES include/cchp.h "${CMAKE_CURRENT_BINARY_DIR}/cchp.def" DESTINATION include)
